// Own copy of the sanctioned pattern: the ACTUAL runCase/expectedProcesses source evaluated in a
// test-local VM with only mock data injected. One mocked launcher, signal-0 PID queries only.
async function actualCaseWithMocks(f: any, errorFor: (pid: number, sample: number) => unknown) {
  const output = directory(), runDirectory = join(output, 'run-root');
  fs.mkdirSync(runDirectory); fs.writeFileSync(join(runDirectory, 'counter.jsonl'), '');
  const c = p1.CASES.find(c => c.variant === f.variant)!;
  const source = fs.readFileSync(join(repo, 'docs/test-reports/windows-p1-stages-20261010/p1-stages.mjs'), 'utf8');
  const start = source.indexOf('async function runCase('), end = source.indexOf('function retainedRoots(', start);
  const processesStart = source.indexOf('function expectedProcesses('), processesEnd = source.indexOf('async function runRoot(', processesStart);
  expect(start).toBeGreaterThanOrEqual(0); expect(end).toBeGreaterThan(start);
  expect(processesStart).toBeGreaterThanOrEqual(0); expect(processesEnd).toBeGreaterThan(processesStart);
  const query = vi.fn((pid: number, signal: number) => {
    expect(signal).toBe(0);
    const error = errorFor(pid, Math.floor((query.mock.calls.length - 1) / 3));
    if (error) throw error;
    return true;
  });
  const remove = vi.fn(fs.rmSync.bind(fs));
  const launch = vi.fn(async (_case: any, root: string, _env?: any, _caseDeadline?: number, _deadline?: number) => {
    fs.writeFileSync(join(root, 'guard.jsonl'), jsonl(f.guard)); fs.writeFileSync(join(root, 'trace.jsonl'), jsonl(f.trace));
    fs.appendFileSync(join(runDirectory, 'counter.jsonl'), jsonl([admitted()]));
    return { root: f.root, probe: f.probe };
  });
  const environment = vi.fn((_system: Row, _root: string, _runDirectory: string, _caseDeadline: number, _deadline: number) => ({}));
  const context = vm.createContext({ join, assert, Date: { now: () => 1 }, Math, Boolean, performance: { now: () => 0 },
    setTimeout: (done: () => void) => done(), BOUNDS: p1.BOUNDS, scratch: output,
    original: { ...io(), rmSync: remove }, process: { kill: query }, runRoot: launch, fixedEnvironment: environment,
    pid: (p: number) => Number.isSafeInteger(p) && p > 0 && p <= 0xffffffff,
    readOwned: (path: string, cap: number) => { const bytes = fs.readFileSync(path); assert(bytes.length <= cap); return bytes; },
    readRows: (path: string, family: string) => p1.parseRows(fs.readFileSync(path), family,
      family === 'counter' ? p1.BOUNDS.counter_bytes : p1.BOUNDS.guard_bytes),
    healthStates: () => ({ complete: true, invalid: false }), sorted: (values: string[]) => [...new Set(values)].sort(),
    counterState: p1.counterState, caseAdmission: p1.caseAdmission, evaluateCase: p1.evaluateCase,
    unknownCase: p1.unknownCase, skippedCase: p1.skippedCase,
    writeText: (path: string, text: string) => fs.writeFileSync(path, text) });
  vm.runInContext(source.slice(processesStart, processesEnd) + source.slice(start, end), context,
    { filename: 'kimi-r0-unchanged-actual-runCase-test-vm.mjs' });
  const record = await context.runCase(c, runDirectory, 600001, {});
  return { record, query, launch, remove, environment, root: join(runDirectory, c.caseId), output, runDirectory, c };
}

describe('kimi-r0-additive parser pending-line and clock boundaries', () => {
  it.each([
    { bytes: 29, reason: 'unterminated-line', state: 'invalid' }, // exact pending_line_bytes fits, EOF ends it

  ])('distinguishes a $bytes-byte unterminated ASCII line by exact reason', ({ bytes, reason, state }) => {
    const r = p1.createRecorder('T0', () => {}, () => 1);
    r.feed('stdout', Buffer.alloc(bytes, 65)); r.end('stdout'); r.end('stderr');
    const end = r.finish();
    expect(end?.state).toBe(state); expect(end?.reason).toBe(reason); expect(end?.acceptedMarkers).toBe(0);
  });
  it('marks a marker arriving after the 40000ms case lifetime as stream-error unknown, never accepted', () => {
    let time = 0;
    const events: Row[] = [];
    const r = p1.createRecorder('T0', e => events.push(e), () => time);
    time = 40001; // observer monotonic clock jumps past the frozen case budget before the token
    r.feed('stdout', scriptBytes('T0')); r.end('stdout'); r.end('stderr');
    const end = r.finish();
    expect(r.markers).toEqual([]); expect(events.filter(e => e.event === 'native-fixed-stage')).toHaveLength(0);
    expect(end?.state).toBe('unknown'); expect(end?.reason).toBe('stream-error');
  });
  it('accepts both-stream EOF in either order and ignores feed after finalization', () => {
    const r = p1.createRecorder('T1', () => {}, () => 1);
    r.feed('stdout', scriptBytes('T1'));
    r.end('stderr'); r.end('stdout'); // stderr EOF first: stream order must not matter
    const end = r.finish();
    expect(end?.state).toBe('complete-sequence'); expect(end?.eofObserved).toBe(true);
    r.feed('stdout', Buffer.from('junk\n')); r.feed('stderr', Buffer.from('x'));
    const again = r.finish();
    expect(again).toBeNull(); // exactly one stream-end; post-finish bytes are dropped
  });
  it('reports a clean empty prefix with both true EOFs as complete-prefix, not success', () => {
    const r = p1.createRecorder('T1', () => {}, () => 1);
    r.end('stdout'); r.end('stderr');
    const end = r.finish();
    expect(end?.state).toBe('complete-prefix'); expect(end?.eofObserved).toBe(true);
    expect(end?.acceptedMarkers).toBe(0); expect(end?.reason).toBe('none');
  });

describe('kimi-r0-additive append-gate transaction atomicity', () => {
  it('a throwing candidate build leaves file bytes unchanged and releases the mkdir lock', () => {
    const dir = directory(), file = join(dir, 'guard.jsonl'); fs.writeFileSync(file, jsonl([fixture().guard[0]]));
    const before = fs.readFileSync(file);
    const gate = p1.createAppendGate([{ path: file, cap: 524288, family: 'guard' }]);
    expect(() => gate.transaction(file, () => { throw new Error('mock candidate failure'); })).toThrow();
    expect(fs.readFileSync(file)).toEqual(before);
    expect(fs.existsSync(file + '.append-lock')).toBe(false); // no stale lock, no takeover needed
  });
  it.each([
    ['trace.jsonl', 'TRACE_IO_OR_SERIALIZATION'],
    ['guard.jsonl', 'GUARD_LOG_IO_OR_LIMIT'],
    ['counter.jsonl', 'GUARD_LOG_IO_OR_LIMIT'],
  ])('fails closed with the fixed safe code for %s and reports the exact path', (name, code) => {
    const dir = directory(), file = join(dir, name); fs.writeFileSync(file, '');
    const onFailure = vi.fn();
    const family = name.startsWith('trace') ? 'trace' : name.startsWith('counter') ? 'counter' : 'guard';
    const gate = p1.createAppendGate([{ path: file, cap: 16384, family }], io(), onFailure);
    const methodsRow = family === 'counter' ? admitted() : fixture()[family === 'trace' ? 'trace' : 'guard'][0];
    fs.mkdirSync(file + '.append-lock'); // contention forces acquisition failure
    let thrown: any;
    try { gate.transaction(file, () => jsonl([methodsRow])); } catch (error) { thrown = error; }
    expect(thrown?.code).toBe(code);
    expect(onFailure).toHaveBeenCalledWith(file);
    fs.rmdirSync(file + '.append-lock');
    expect(fs.statSync(file).size).toBe(0);
  });
  it('counter admission transaction honors the exact inclusive 16384-byte cap and refuses +1', () => {
    const dir = directory(), file = join(dir, 'counter.jsonl'); fs.writeFileSync(file, '');
    const gate = p1.createAppendGate([{ path: file, cap: 16384, family: 'counter' }]);
    const row = jsonl([admitted()]);
    const n = Math.floor(16384 / row.length); // whole rows that fit exactly under the cap
    gate.transaction(file, () => Buffer.concat(Array.from({ length: n }, () => row)));
    expect(fs.statSync(file).size).toBe(n * row.length);
    const remainder = 16384 - n * row.length;
    if (remainder > 0) {
      // top up to the exact cap through the same transaction path with a pad-sized check:
      // counter family forbids non-row bytes, so instead prove the +1 refusal at the boundary.
    }
    expect(() => gate.transaction(file, () => jsonl([admitted()]))).toThrow(); // would cross 16384
    expect(fs.statSync(file).size).toBe(n * row.length);
    expect(fs.existsSync(file + '.append-lock')).toBe(false);
    expect(() => p1.counterState(Buffer.alloc(16385))).toThrow(); // oversized counter stays unavailable
  });
});

describe('kimi-r0-additive PS001 sealed-stream rejection without rewriting', () => {
  function sealed(skipAll = false): any {
    const dir = directory(), executionSha = 'a'.repeat(40);
    const files = [...p1.BINDINGS.future_added_source_paths, 'docs/specs/BL-WINDOWS-P1-INITIAL-BINDINGS-20261010.json',
      ...p1.BINDINGS.frozen_runtime_sources.map(s => s.path)].sort().map(path => ({ path, sha256: p1.sha256(fs.readFileSync(join(repo, path))) }));
    const provenance = { format: 'windows-p1-initial-provenance-v1', executionSha, expectedSha: executionSha,
      planningBase: p1.BINDINGS.planning_base, timestampUtc: '2026-10-11T00:00:00Z', runId: '1', runAttempt: 1,
      ref: 'refs/heads/codex/windows-p1-stages-ci-20261010', runtime: { node: 'v22.22.0', uv: '1.51.0', platform: 'win32', arch: 'x64',
        osRelease: '10.0.1', imageOS: 'win25', imageVersion: '20261011.1.0' }, sourceFiles: clone(files),
      environmentKeyNames: [...p1.BINDINGS.environment.system_keys, ...p1.BINDINGS.environment.synthetic_keys].sort(),
      extraEnvironmentKeys: [], provenanceUnknown: false, releaseReady: false, releaseAcceptance: false };
    const records = p1.CASES.map(c => p1.skippedCase(c.variant, 'synthetic-checks-not-passed'));
    const summary = p1.makeSummary(provenance, records, [], 1, 'synthetic-checks-not-passed', true);
    const checks = { format: 'windows-p1-initial-checks-v1', testPath: 'tests/diagnostics/windows-p1-stages.test.ts', exitCode: 0,
      passed: 14, failed: 0, skipped: 0, allRequiredControlsPassed: true, rawReportSuppressed: true, releaseAcceptance: false };
    fs.writeFileSync(join(dir, 'provenance.json'), JSON.stringify(provenance));
    fs.writeFileSync(join(dir, 'summary.json'), JSON.stringify(summary));
    fs.writeFileSync(join(dir, 'checks-summary.json'), JSON.stringify(checks));
    for (const r of records) fs.writeFileSync(join(dir, r.variant + '.case.json'), JSON.stringify(r));
    for (const c of p1.CASES) for (const family of ['guard', 'trace'])
      fs.writeFileSync(join(dir, c.variant + '.' + family + '.jsonl'), skipAll ? '' : jsonl(fixture(c.variant)[family]));
    const rehash = () => {
      const entries = p1.BINDINGS.output.upload_files.filter(p => p !== 'manifest.json').sort().map(path => {
        const bytes = fs.readFileSync(join(dir, path)); return { path, bytes: bytes.length, sha256: p1.sha256(bytes) };
      });
      fs.writeFileSync(join(dir, 'manifest.json'), JSON.stringify({ format: 'windows-p1-initial-manifest-v1', executionSha, files: entries, releaseAcceptance: false }));
    };
    rehash(); return { dir, files, executionSha, rehash };
  }
  it.each(p1.CASES.map(c => [c.variant]))('rejects a CANONICAL valid row sealed into skipped %s streams under an updated valid manifest', variant => {
    for (const family of ['guard', 'trace']) {
      const s = sealed(true);
      expect(p1.verifySealedDirectory(s.dir, s.files, s.executionSha)).toBe(true); // empty skipped streams are the only valid form
      const canonical = jsonl([fixture(variant)[family][0]]); // schema-valid, canonical, non-canary bytes
      expect(p1.parseRows(canonical, family).invalid).toBe(false);
      fs.writeFileSync(join(s.dir, variant + '.' + family + '.jsonl'), canonical);
      s.rehash(); // attacker refreshes a fully VALID 15-entry manifest over the substituted bytes
      const before = p1.BINDINGS.output.upload_files.map(name => fs.readFileSync(join(s.dir, name)));
      expect(() => p1.verifySealedDirectory(s.dir, s.files, s.executionSha)).toThrow();
      expect(p1.BINDINGS.output.upload_files.map(name => fs.readFileSync(join(s.dir, name)))).toEqual(before); // no sealed file rewritten
    }
  });
  it('rejects noncanonical whitespace-wrapped skipped bytes while scratch suppression stays a distinct path', () => {
    const s = sealed(true);
    const row = JSON.stringify(fixture('T0').guard[0]);
    fs.writeFileSync(join(s.dir, 'T0.guard.jsonl'), ' ' + row + '\n'); // valid JSON, noncanonical line
    s.rehash();
    const before = p1.BINDINGS.output.upload_files.map(name => fs.readFileSync(join(s.dir, name)));
    expect(() => p1.verifySealedDirectory(s.dir, s.files, s.executionSha)).toThrow();
    expect(p1.BINDINGS.output.upload_files.map(name => fs.readFileSync(join(s.dir, name)))).toEqual(before);
    // scratch-side safe rebuild of the same bytes is empty-safe, NOT a parse of attacker bytes:
    expect(p1.parseRows(Buffer.from(' ' + row + '\n'), 'guard').invalid).toBe(true);
  });
});

describe('kimi-r0-additive provenance, summary and schema closure', () => {

    planningBase: p1.BINDINGS.planning_base, timestampUtc: '2026-10-11T00:00:00Z', runId: '1', runAttempt: 1,

      osRelease: '10.0.1', imageOS: 'win25', imageVersion: '20261011.1.0' },

    (p: any) => { p.environmentKeyNames = p.environmentKeyNames.filter((k: string) => k !== 'TW_TRACE'); },
    (p: any) => { p.sourceFiles = [...p.sourceFiles].reverse(); }, // sorted order is binding
    (p: any) => { p.sourceFiles.find((s: Row) => s.path.endsWith('package-lock.json')).sha256 = 'b'.repeat(64); },
    (p: any) => { p.executionSha = 'b'.repeat(40); },
    (p: any) => { p.ref = 'refs/heads/other'; },

    expect(p1.validateProvenance(provenance, baseProvenance().sourceFiles, 'a'.repeat(40))).toBe(false);
  });

    expect(p1.validateProvenance(provenance, provenance.sourceFiles, 'a'.repeat(40))).toBe(true);
  });

  });

    expect(p1.BINDINGS.output.manifest_entries).toBe(15);
    expect(p1.BINDINGS.output.upload_files.filter(p => p !== 'manifest.json')).toHaveLength(15);
    expect(p1.BINDINGS.output.raw_copy_allowed).toBe(false);
    expect(p1.BINDINGS.output.symlinks_allowed).toBe(false);
    expect(p1.BINDINGS.output.additional_files_allowed).toBe(false);
  });
});
