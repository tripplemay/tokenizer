function git(args: string[], cwd = repo, env: NodeJS.ProcessEnv = gitEnv) {
  const r = cp.spawnSync('git', args, { cwd, env, encoding: 'buffer', timeout: 30000, maxBuffer: 64 * 1024 * 1024 });
  if (r.error) throw r.error;
  return r;
}
const gitShow = (ref: string, path: string) => {
  const r = git(['-c', 'core.fileMode=false', 'show', `${ref}:${path}`]);
  expect(r.status).toBe(0);
  return r.stdout;
};

describe('r1-exact-byte-projections-from-source888', () => {
  const proposal = JSON.parse(fs.readFileSync(join(repo, 'docs/test-reports/BL-WINDOWS-P1-CHECKOUT-plan-20261011/proposal.json'), 'utf8'));
  it('workflow = source888 plus exactly one replacement (four added env lines under checkout)', () => {
    const before = gitShow(SOURCE888, WORKFLOW).toString();
    const current = fs.readFileSync(join(repo, WORKFLOW), 'utf8');
    const delta = proposal.selected_solution.workflow_delta;
    expect(before.split(delta.before).length - 1).toBe(1);
    expect(before.replace(delta.before, delta.after)).toBe(current);
  });
  it('bindings JSON = source888 with exactly the five declared unique replacements', () => {
    let render = gitShow(SOURCE888, BINDING_JSON).toString();
    const current = fs.readFileSync(join(repo, BINDING_JSON), 'utf8');
    for (const rep of proposal.contract_revision_required.binding_json_replacements) {
      expect(render.split(rep.before).length - 1, rep.before.slice(0, 40)).toBe(1);
      expect(current.split(rep.after).length - 1, rep.after.slice(0, 40)).toBe(1);
      render = render.replace(rep.before, rep.after);
    }
    expect(render).toBe(current);
  });
  it('bindings JSON partial application (stale revision) never reproduces the candidate', () => {
    const old = gitShow(SOURCE888, BINDING_JSON).toString();
    const reps = proposal.contract_revision_required.binding_json_replacements;
    const current = fs.readFileSync(join(repo, BINDING_JSON), 'utf8');
    for (let skip = 0; skip < reps.length; skip++) {
      let render = old;
      reps.forEach((rep: any, i: number) => { if (i !== skip) render = render.replace(rep.before, rep.after); });
      expect(render).not.toBe(current);
    }
  });
  it('bindings MD = source888 bytes plus exactly the declared appendix', () => {
    const old = gitShow(SOURCE888, BINDING_MD);
    const current = fs.readFileSync(join(repo, BINDING_MD));
    expect(current.subarray(0, old.length).equals(old)).toBe(true);
    expect(current.subarray(old.length).toString('utf8')).toBe(proposal.contract_revision_required.binding_md_append);
  });
  it('CLI masked for exactly the two identity literals is byte-identical to source888', () => {
    const old = gitShow(SOURCE888, CLI).toString();
    const current = fs.readFileSync(join(repo, CLI), 'utf8');
    const masked = current.replace(HASHES[BINDING_JSON], OLD_BINDING_HASH).replace(HASHES[BINDING_MD], OLD_MD_HASH);
    expect(masked).toBe(old);
    expect(current.replace(HASHES[BINDING_JSON], OLD_BINDING_HASH)).not.toBe(old);
    expect(current.replace(HASHES[BINDING_MD], OLD_MD_HASH)).not.toBe(old);
    expect(masked + '\n').not.toBe(old);
    expect(old).toContain(`'${OLD_BINDING_HASH}'`);
    expect(old).toContain(`'${OLD_MD_HASH}'`);
  });
});

    const src = fs.readFileSync(join(repo, CLI), 'utf8');
    expect(src).toContain(`const bindingHash = '${sha256(fs.readFileSync(join(repo, BINDING_JSON)))}';`);
    expect(src).toContain(`const mdHash = '${sha256(fs.readFileSync(join(repo, BINDING_MD)))}';`);
  });
  it('stale literals bind only stale documents: old hashes do not match candidate documents', () => {
    expect(sha256(fs.readFileSync(join(repo, BINDING_JSON)))).not.toBe(OLD_BINDING_HASH);
    expect(sha256(fs.readFileSync(join(repo, BINDING_MD)))).not.toBe(OLD_MD_HASH);
    expect(sha256(gitShow(SOURCE888, BINDING_JSON))).toBe(OLD_BINDING_HASH);
    expect(sha256(gitShow(SOURCE888, BINDING_MD))).toBe(OLD_MD_HASH);
  });
  const mirrorFiles = [BINDING_JSON, BINDING_MD, CLI,
    'docs/test-reports/windows-release-diagnosis-20261008/native-runtime-parity.mjs',
    'docs/test-reports/windows-release-diagnosis-20261008/trace-preload.mjs',
    'docs/specs/BL-WINDOWS-P1-STAGES-20261009.md',
    'docs/test-reports/BL-WINDOWS-P1-STAGES-plan-20261009/plan.json',
    'docs/test-reports/BL-WINDOWS-P1-STAGES-scope-20261009/verdict.json',
    'src/cli/bounded-subprocess.ts', 'src/cli/bounded-subprocess-worker.mjs', 'src/cli/replay.ts',
    '.github/workflows/windows-release-diagnostics.yml', 'package-lock.json'];
  function makeMirror(mutate: (mirror: string) => void): string {
    const mirror = fs.mkdtempSync(join(owned, 'mirror-'));
    for (const rel of mirrorFiles) {
      const dst = join(mirror, rel);
      fs.mkdirSync(dirname(dst), { recursive: true });
      fs.copyFileSync(join(repo, rel), dst);
    }
    mutate(mirror);
    return mirror;
  }
  function importAndValidate(mirror: string) {
    const url = pathToFileURL(join(mirror, CLI)).href;
    const script = `const m = await import(${JSON.stringify(url)}); m.validateBindings(); console.log('VALIDATE_OK');`;
    return cp.spawnSync(process.execPath, ['--input-type=module', '-e', script],
      { env: { HOME: join(mirror, 'home'), TMPDIR: owned, PATH: '/usr/bin:/bin', NODE_ENV: 'test' }, encoding: 'utf8', timeout: 30000 });
  }
  it('baseline mirror of the real candidate imports and validateBindings passes', () => {
    const r = importAndValidate(makeMirror(() => {}));
    expect(r.status).toBe(0);
    expect(r.stdout).toContain('VALIDATE_OK');
  });
  const staleCases: Array<[string, (mirror: string) => void, string]> = [
    ['json-one-byte-mutation', m => {
      const p = join(m, BINDING_JSON);
      fs.writeFileSync(p, fs.readFileSync(p, 'utf8').replace('"revision": 2', '"revision": 3'));

    ['stale-cli-old-source-relabelled', m => {
      fs.writeFileSync(join(m, CLI), gitShow(SOURCE888, CLI));

      fs.writeFileSync(p, fs.readFileSync(p, 'utf8').replace(HASHES[BINDING_JSON], OLD_BINDING_HASH));
    }, 'binding identity'],
    ['stale-md-literal-only', m => {
      const p = join(m, CLI);
      fs.writeFileSync(p, fs.readFileSync(p, 'utf8').replace(HASHES[BINDING_MD], OLD_MD_HASH));
    }, 'AssertionError'],
    ['md-one-byte-mutation', m => {
      const p = join(m, BINDING_MD);
      fs.writeFileSync(p, fs.readFileSync(p, 'utf8') + ' ');
    }, 'AssertionError']
  ];
  it.each(staleCases)('negative %s is rejected by unchanged import/validateBindings', (_name, mutate, marker) => {
    const r = importAndValidate(makeMirror(mutate));
    expect(r.status).not.toBe(0);
    expect(r.stdout).not.toContain('VALIDATE_OK');
    expect(r.stderr).toContain(marker);
  });
});

    GIT_CONFIG_NOSYSTEM: '1', GIT_CONFIG_GLOBAL: globalConfig, XDG_CONFIG_HOME: xdg, GIT_TERMINAL_PROMPT: '0',
    NODE_ENV: 'test'
  };
  const withCheckoutKeys = { ...childEnv, GIT_CONFIG_COUNT: '1', GIT_CONFIG_KEY_0: 'core.longpaths', GIT_CONFIG_VALUE_0: 'true' };
  const configBytes = () => fs.readFileSync(join(fixture, '.git', 'config'));
  it('command-scope env injects core.longpaths=true for this Git child only, with no persistence', () => {
    expect(git(['init', '-q'], fixture, childEnv).status).toBe(0);
    const before = configBytes();
    const injected = git(['config', '--get', 'core.longpaths'], fixture, withCheckoutKeys);
    expect(injected.status).toBe(0);
    expect(injected.stdout.toString()).toBe('true\n');
    const origin = git(['config', '--show-origin', '--get', 'core.longpaths'], fixture, withCheckoutKeys);
    expect(origin.status).toBe(0);
    expect(origin.stdout.toString()).toContain('command line');
    const sibling = git(['config', '--get', 'core.longpaths'], fixture, childEnv);
    expect(sibling.status).not.toBe(0);
    expect(sibling.stdout.toString()).toBe('');
    expect(fs.readFileSync(globalConfig).toString()).toBe('');
    expect(configBytes().equals(before)).toBe(true);
    expect(configBytes().toString()).not.toContain('longpaths');
    expect(fs.readdirSync(xdg)).toEqual([]);
  });
});
