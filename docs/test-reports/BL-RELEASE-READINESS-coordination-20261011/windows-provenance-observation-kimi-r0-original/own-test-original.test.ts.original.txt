function build(overrides: Record<string, any> = {}, opts: { rebuildSource?: string } = {}) {
  const stderr: string[] = [], stdout: string[] = [];
  const calls: Record<string, number> = { schema: 0, stringify: 0, parse: 0, same: 0, sorted: 0, sourceFiles: 0, writeJson: 0 };
  const order: string[] = [];
  const sameArgs: any[][] = [];
  const realJson = JSON;
  const sandbox: Record<string, any> = {
    schemaResult: true,
    schemaMatches: (_s: any, _v: any) => { calls.schema++; order.push('schema'); return sandbox.schemaResult; },
    defs: { provenance: { stub: true } },
    JSON: {
      stringify: (v: any) => { calls.stringify++; order.push('stringify'); return realJson.stringify(v); },
      parse: (v: string) => { calls.parse++; order.push('parse'); return realJson.parse(v); }
    },
    same: (a: any, b: any) => { calls.same++; sameArgs.push([a, b]); return realJson.stringify(a) === realJson.stringify(b); },
    sorted: (vs: string[]) => { calls.sorted++; return [...new Set(vs)].sort(); },
    BINDINGS: {
      planning_base: 'base', future_added_source_paths: ['fut.ts'],
      frozen_runtime_sources: [{ path: 'fr.ts', sha256: 'hfr' }],
      environment: { system_keys: ['SystemRoot', 'SYS_A'], synthetic_keys: ['SYN_A'] }
    },
    bindingPath: 'binding.json', bindingHash: 'hb',
    assert: Object.assign((cond: any) => { if (!cond) throw new Error('assertion'); }, {
      equal: (a: any, b: any) => { if (a !== b) throw new Error('assertion.equal'); }
    }),
    assertWindowsRunner: () => {}, validateBindings: () => {}, checkoutSha: () => A40,
    systemEnvironment: () => ({ SystemRoot: 'C:\\Windows' }), safeDirectory: () => {},
    original: { mkdirSync: () => {} },
    sourceFiles: () => { calls.sourceFiles++; return realJson.parse(realJson.stringify(VALID_FILES)); },
    writeJson: () => { calls.writeJson++; },
    release: () => '10.0.22631', scratch: 'scratch-dir',
    join: (...parts: string[]) => parts.join('/'), dirname: (p: string) => p,
    process: {
      argv: ['node', 'p1-stages.mjs', '--prepare'],
      env: { GITHUB_EVENT_NAME: 'push', GITHUB_REF: 'refs/heads/main', GITHUB_SHA: A40, GITHUB_RUN_ID: '1', GITHUB_RUN_ATTEMPT: '1', ImageOS: 'win', ImageVersion: '1' },
      version: 'v22.22.0', versions: { uv: '1.51.0' }, platform: 'win32', arch: 'x64',
      stderr: { write: (s: any) => { stderr.push(String(s)); return true; } },
      stdout: { write: (s: any) => { stdout.push(String(s)); return true; } },
      exitCode: undefined as any
    },
    runParent: async () => {}, runProbe: async () => {}, seal: () => {}, verifyUpload: () => {},
    assertWindowsRunnerForProbe: () => {},
    CASES: [{ variant: 'T1', caseId: 'case-t1', scriptSha256: 'sh', sequence: [] }],
    caseFor: (v: string) => sandbox.CASES.find((c: any) => c.variant === v),
    unknownNative: () => ({ unknown: true })
  };
  Object.assign(sandbox, overrides);
  const ctx = vm.createContext(sandbox);
  vm.runInContext(
    declSlice + (opts.rebuildSource ?? rebuildSlice) + validateSlice + prepareSlice +
    '\nglobalThis.__dispatch = async () => { ' + dispatchSlice + '\n};\n' +
    'globalThis.__terminal = () => { const args = process.argv.slice(2); try { throw 0; } ' + catchSlice + '\n};\n' +
    'globalThis.__state = () => ({ phase: prepareFailurePhase, active: prepareProvenanceObservationActive, reason: prepareProvenanceReason });\n' +
    'globalThis.__set = (p, a, r) => { prepareFailurePhase = p; prepareProvenanceObservationActive = a; prepareProvenanceReason = r; };\n' +
    'globalThis.__fns = { rebuild, validateProvenance, prepare };',
    ctx);
  const run = (code: string) => vm.runInContext(code, ctx);
  return {
    sandbox, stderr, stdout, calls, order, sameArgs,
    dispatch: async () => { await run('__dispatch()'); },
    terminal: () => run('__terminal()'),
    state: () => run('__state()'),
    set: (p: any, a: boolean, r: string) => run(`__set(${realJson.stringify(p)},${a},${realJson.stringify(r)})`),
    fns: () => run('__fns'),
    stderrText: () => stderr.join(''),
    stdoutText: () => stdout.join('')
  };
}
type H = ReturnType<typeof build>;

const expectTerminal = (h: H, phase: string | null, reason: string | null) => {
  const expected = GENERIC + (phase ? P_LINE(phase) : '') + (reason ? `${reason}\n` : '');
  expect(h.stderrText()).toBe(expected);
  expect(h.stdoutText()).toBe('');
  expect(h.sandbox.process.exitCode).toBe(1);
};
const validateAtP11 = (h: H, value: any, files: any, sha: string) => {
  h.set('P11', true, 'V00');
  let thrown: any = null, result: any = null;
  try { result = h.fns().validateProvenance(value, files, sha); } catch (e) { thrown = e; }
  h.terminal();
  return { thrown, result };
};

describe('rebuild slice: schema/clone stage classification with exact call order and counts', () => {
  it('schema true returns deep clone, calls schema then stringify then parse exactly once, no latch', () => {
    const h = build();
    h.set('P11', true, 'V00');
    const value = validProvenance();
    const out = h.fns().rebuild('provenance', value);
    expect(out).toEqual(value);
    expect(out).not.toBe(value);
    expect(h.calls).toMatchObject({ schema: 1, stringify: 1, parse: 1 });
    expect(h.order).toEqual(['schema', 'stringify', 'parse']);
    expect(h.state().reason).toBe('V00'); // rebuild alone never emits/latches success
  });
  it('schema false returns null and latches V02, clone never attempted', () => {
    const h = build({ schemaResult: false });
    h.set('P11', true, 'V00');
    expect(h.fns().rebuild('provenance', validProvenance())).toBe(null);
    expect(h.state().reason).toBe('V02');
    expect(h.calls).toMatchObject({ schema: 1, stringify: 0, parse: 0 });
  });
  it('caught schema escape returns null and latches V03 without inspecting the thrown value', () => {
    const err = new Error('hidden');
    Object.defineProperty(err, 'message', { get() { throw new Error('MESSAGE-READ-CANARY'); } });
    let schemaCalls = 0;
    const h = build({ schemaMatches: () => { schemaCalls++; throw err; } });
    h.set('P11', true, 'V00');
    expect(h.fns().rebuild('provenance', validProvenance())).toBe(null);
    expect(h.state().reason).toBe('V03');
    expect(schemaCalls).toBe(1);
    expect(h.calls).toMatchObject({ stringify: 0, parse: 0 });
  });
  it('caught primitive escape also classifies V03 and never serializes the primitive', () => {
    const h = build({ schemaMatches: () => { throw 'PRIMITIVE-CANARY'; } });
    h.set('P11', true, 'V00');
    expect(h.fns().rebuild('provenance', validProvenance())).toBe(null);
    expect(h.state().reason).toBe('V03');
  });
  it('caught clone escape returns null and latches V04, stringify attempted exactly once', () => {
    let toJson = 0;
    const value = { ...validProvenance(), toJSON() { toJson++; throw new Error('TOJSON-SECRET'); } };
    const h = build();
    h.set('P11', true, 'V00');
    expect(h.fns().rebuild('provenance', value)).toBe(null);
    expect(h.state().reason).toBe('V04');
    expect(h.calls).toMatchObject({ schema: 1, stringify: 1, parse: 0 });
    expect(toJson).toBe(1); // original single stringify behaviour preserved, not suppressed/duplicated
  });
  it('inactive observer (not exact prepare P11, or non-provenance name) changes nothing and emits nothing', () => {
    for (const [phase, active, name] of [['P11', false, 'provenance'], ['P12', true, 'provenance'], [null, true, 'provenance'], ['P11', true, 'other']] as any[]) {
      const h = build({ schemaResult: false });
      h.set(phase, active, 'V00');
      expect(h.fns().rebuild(name, validProvenance())).toBe(null); // identical original behaviour
      expect(h.state().reason).toBe('V00');
      expect(h.calls.schema).toBe(1);
    }
  });
});

describe('validator call order, arguments, short-circuit and property reads', () => {
  it('happy path: true result, exact same() argument sequence, sorted once, no reason emission on success path', () => {
    const h = build();
    h.set('P11', true, 'V00');
    const value = validProvenance(), files = JSON.parse(JSON.stringify(VALID_FILES));
    expect(h.fns().validateProvenance(value, files, A40)).toBe(true);
    expect(h.state().reason).toBe('V17');
    expect(h.calls).toMatchObject({ schema: 1, stringify: 1, parse: 1, same: 3, sorted: 1 });
    expect(h.sameArgs[0]).toEqual([value.sourceFiles, files]);
    expect(h.sameArgs[1]).toEqual([['binding.json', 'fr.ts', 'fut.ts'], ['binding.json', 'fr.ts', 'fut.ts']]);
    expect(h.sameArgs[2]).toEqual([['SYN_A', 'SystemRoot'], ['SYN_A', 'SystemRoot']]);
  });
  it('short-circuit: second predicate true skips the third and same() entirely', () => {
    const h = build();
    const value = { ...validProvenance(), expectedSha: 'c'.repeat(40) };
    expect(h.fns().validateProvenance(value, VALID_FILES, A40)).toBe(false);
    expect(h.calls.same).toBe(0);
  });
  it('reads only the original five properties; canary members are never read', () => {
    const reads: string[] = [];
    const data = validProvenance();
    const prox = new Proxy(data, { get(t, k) { if (typeof k === 'string') reads.push(k); return (t as any)[k]; } });
    const h = build({ __prox: prox }, { rebuildSource: 'function rebuild() { return globalThis.__prox; }' });
    h.set('P11', true, 'V00');
    expect(h.fns().validateProvenance(prox, VALID_FILES, A40)).toBe(true);
    const allowed = new Set(['executionSha', 'expectedSha', 'sourceFiles', 'environmentKeyNames', 'extraEnvironmentKeys']);
    expect(reads.every(k => allowed.has(k))).toBe(true);
    expect(reads).not.toContain('secret');
    expect(reads).not.toContain('runId');
    expect(reads).not.toContain('format');
    expect(reads).toContain('executionSha');
  });
  it('original clone reads each enumerable getter exactly once, observer adds no further reads', () => {
    let secretReads = 0;
    const value = validProvenance();
    Object.defineProperty(value, 'secret', { enumerable: true, get() { secretReads++; return 'S'; } });
    const h = build();
    h.set('P11', true, 'V00');
    expect(h.fns().validateProvenance(value, VALID_FILES, A40)).toBe(true);
    expect(secretReads).toBe(1); // the single original JSON clone read
  });
});

describe('prepare dispatch: success silence/reset, old P00-P12 output and generic exit', () => {
  it('successful prepare: zero stdout/stderr, no exit code, both sourceFiles passes, observer deactivated', async () => {
    const h = build();
    await h.dispatch();
    expect(h.stderrText()).toBe('');
    expect(h.stdoutText()).toBe('');
    expect(h.sandbox.process.exitCode).toBe(undefined);
    expect(h.state()).toEqual({ phase: null, active: false, reason: 'V17' });
    expect(h.calls.sourceFiles).toBe(2); // both original passes retained
    expect(h.calls.writeJson).toBe(1);
    expect(h.calls.schema).toBe(1);
  });
  it('activation reset: a failed run leaves state, next --prepare arm resets V00 before prepare', async () => {
    const h = build({ schemaResult: false });
    await h.dispatch();
    expect(h.state()).toEqual({ phase: 'P11', active: true, reason: 'V02' });
    h.sandbox.schemaResult = true;
    h.stderr.length = 0; h.stdout.length = 0; h.sandbox.process.exitCode = undefined;
    await h.dispatch();
    expect(h.state()).toEqual({ phase: null, active: false, reason: 'V17' });
    expect(h.stderrText()).toBe(''); // successful rerun stays silent
  });
  it.each([
    ['P01', { assertWindowsRunner: () => { throw new Error('x'); } }],
    ['P02', { validateBindings: () => { throw new Error('x'); } }],
    ['P03', { checkoutSha: () => { throw new Error('x'); } }],
    ['P04', { checkoutSha: () => 'b'.repeat(40) }],
    ['P05', { systemEnvironment: () => { throw new Error('x'); } }],
    ['P06', { systemEnvironment: () => ({ SystemRoot: 'bad' }) }],
    ['P07', { safeDirectory: () => { throw new Error('x'); } }],
    ['P08', { original: { mkdirSync: () => { throw new Error('x'); } } }],
    ['P09', { sourceFiles: () => { throw new Error('x'); } }],
    ['P10', { sourceFiles: (() => { let n = 0; return () => { n++; if (n === 2) throw new Error('x'); return JSON.parse(JSON.stringify(VALID_FILES)); }; })() }],
    ['P12', { writeJson: () => { throw new Error('x'); } }]
  ])('failure at %s emits only generic+phase (80 bytes), no reason, exit 1', async (phase, overrides) => {
    const h = build(overrides as any);
    await h.dispatch();
    expectTerminal(h, phase as string, null);
    expect(Buffer.byteLength(h.stderrText())).toBe(80);
    expect(h.stderr.length).toBe(2);
    expect(h.state().phase).toBe(phase);
  });
  it('nonmember phase falls back to fixed P00 line; null phase emits generic only (51 bytes)', () => {
    const h = build();
    h.set('P99', false, 'V00');
    h.terminal();
    expectTerminal(h, 'P00', null);
    const h2 = build();
    h2.set(null, false, 'V00');
    h2.terminal();
    expectTerminal(h2, null, null);
    expect(Buffer.byteLength(h2.stderrText())).toBe(51);
  });
  it('non-P11 phases with active observation never emit a reason (P12 writeJson/rebuild leave observer unchanged)', async () => {
    const h = build({ writeJson: () => { throw new Error('x'); } });
    await h.dispatch();
    expect(h.state()).toEqual({ phase: 'P12', active: true, reason: 'V17' });
    expectTerminal(h, 'P12', null);
    expect(Buffer.byteLength(h.stderrText())).toBe(80);
    const h2 = build();
    h2.set('P12', true, 'V17');
    expect(h2.fns().rebuild('provenance', validProvenance())).toEqual(validProvenance()); // P12 clone path identical
    expect(h2.state().reason).toBe('V17'); // untouched by rebuild outside P11
    h2.terminal();
    expectTerminal(h2, 'P12', null);
  });
  it('other modes stay silent: --run, --seal, --verify-upload, unknown mode, probe success and probe failure', async () => {
    for (const argv of [['node', 'cli', '--run'], ['node', 'cli', '--seal'], ['node', 'cli', '--verify-upload'], ['node', 'cli', '--probe', 'T1']]) {
      const h = build();
      h.sandbox.process.argv = argv;
      await h.dispatch();
      expect(h.stderrText()).toBe('');
      expect(h.state()).toEqual({ phase: null, active: false, reason: 'V00' });
      expect(h.sandbox.process.exitCode).toBe(undefined);
    }
    const unknown = build();
    unknown.sandbox.process.argv = ['node', 'cli', '--bogus'];
    await unknown.dispatch();
    expectTerminal(unknown, null, null);
    const probeFail = build({ runProbe: async () => { throw new Error('probe-x'); } });
    probeFail.sandbox.process.argv = ['node', 'cli', '--probe', 'T1'];
    await probeFail.dispatch();
    expect(probeFail.stderrText()).toBe(''); // probe failure uses original stdout JSON, never the observer
    expect(probeFail.stdoutText()).toContain('windows-p1-initial-probe-v1');
    expect(probeFail.sandbox.process.exitCode).toBe(1);
    expect(probeFail.state()).toEqual({ phase: null, active: false, reason: 'V00' });
  });
});
