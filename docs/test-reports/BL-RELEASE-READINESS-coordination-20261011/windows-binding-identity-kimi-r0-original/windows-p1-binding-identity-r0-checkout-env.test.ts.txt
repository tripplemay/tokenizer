// Strict exact-byte scope oracle for the checkout-step env map (not a YAML parser, not a runtime validator).
function checkoutEnvOracle(text: string): string[] {
  const violations: string[] = [];
  const lines = text.split('\n');
  const checkout = lines.findIndex(l => l.trim() === '- uses: actions/checkout@v4');
  if (checkout < 0) return ['no checkout step'];
  const envAt = lines.findIndex((l, i) => i > checkout && l === '        env:');
  if (envAt !== checkout + 1) violations.push('checkout env not immediately under the checkout step');
  if (JSON.stringify(lines.slice(envAt + 1, envAt + 1 + ENV_LINES.length)) !== JSON.stringify(ENV_LINES)) violations.push('checkout env is not the exact ordered five-line map');
  if (lines[envAt + 1 + ENV_LINES.length] !== '        with:') violations.push('env block not closed by with:');
  if (lines.filter(l => /GIT_CONFIG/.test(l)).length !== 5) violations.push('GIT_CONFIG line count != 5 (leakage or missing)');
  if (/GITHUB_ENV/.test(text)) violations.push('GITHUB_ENV leakage');
  return violations;
}
function bindingOracle(j: any): string[] {
  const v: string[] = [];
  if (j.revision !== 3) v.push('revision');
  if (j.updated_at_utc !== '2026-10-11T16:11:16Z') v.push('updated_at_utc');
  if (j.revision_scope !== SCOPE) v.push('revision_scope');
  if (JSON.stringify(j.workflow?.checkout_env) !== JSON.stringify(FIVE)) v.push('checkout_env');
  if (j.workflow?.checkout_env_scope !== CONFINEMENT) v.push('checkout_env_scope');
  return v;
}

describe('owned disposable Git fixture (local transform evidence, NOT Windows observation)', () => {
  const tmp = process.env.TMPDIR;
  if (!tmp) throw new Error('owned TMPDIR required');
  const root = fs.mkdtempSync(join(tmp, 'wp1cb-r0-git-'));
  const home = join(root, 'home'), xdgtmp = join(root, 'xdg'), repoA = join(root, 'repo-a');
  fs.mkdirSync(home, { recursive: true }); fs.mkdirSync(xdgtmp); fs.mkdirSync(repoA);
  fs.writeFileSync(join(home, 'empty-gitconfig'), '');
  const baseEnv: Record<string, string | undefined> = {
    PATH: process.env.PATH, HOME: home, USERPROFILE: home, XDG_CONFIG_HOME: xdgtmp, TMPDIR: tmp,
    GIT_CONFIG_NOSYSTEM: '1', GIT_CONFIG_GLOBAL: join(home, 'empty-gitconfig'), GIT_TERMINAL_PROMPT: '0',
    GIT_AUTHOR_NAME: 't', GIT_AUTHOR_EMAIL: 't@t', GIT_COMMITTER_NAME: 't', GIT_COMMITTER_EMAIL: 't@t',
  };
  const FIVE_ENV = { GIT_CONFIG_COUNT: '2', GIT_CONFIG_KEY_0: 'core.longpaths', GIT_CONFIG_VALUE_0: 'true', GIT_CONFIG_KEY_1: 'core.autocrlf', GIT_CONFIG_VALUE_1: 'false' };
  const git = (args: string[], env: Record<string, string | undefined> = {}) =>
    cp.execFileSync('git', args, { cwd: repoA, env: { ...baseEnv, ...env } as any });
  const sourceBytes = fs.readFileSync(JS);
  afterAll(() => fs.rmSync(root, { recursive: true, force: true }));
  it('command-scope five keys (longpaths=true autocrlf=false) preserve raw source JSON bytes', () => {
    git(['init', '-q']);
    fs.writeFileSync(join(repoA, 'bindings.json'), sourceBytes);
    git(['-c', 'core.autocrlf=false', 'add', 'bindings.json']);
    git(['-c', 'core.autocrlf=false', 'commit', '-q', '-m', 'fixture']);
    fs.rmSync(join(repoA, 'bindings.json'));
    git(['checkout-index', '-f', '-a'], FIVE_ENV);
    expect(fs.readFileSync(join(repoA, 'bindings.json')).equals(sourceBytes)).toBe(true);
  });
  it('one-variable counterfactual: autocrlf=true produces exactly the LF->CRLF transform', () => {
    fs.rmSync(join(repoA, 'bindings.json'));
    git(['checkout-index', '-f', '-a'], { ...FIVE_ENV, GIT_CONFIG_VALUE_1: 'true' });
    const out = fs.readFileSync(join(repoA, 'bindings.json'));
    const crlf = Buffer.from(sourceBytes.toString('utf8').replace(/\n/g, '\r\n'));
    expect(out.equals(crlf)).toBe(true);
    expect(out.equals(sourceBytes)).toBe(false);
    expect(sha256(out)).toBe(sha256(crlf));
  });
  it('sibling without the five keys shows no persistence and default LF bytes', () => {
    fs.rmSync(join(repoA, 'bindings.json'));
    git(['checkout-index', '-f', '-a']);
    expect(fs.readFileSync(join(repoA, 'bindings.json')).equals(sourceBytes)).toBe(true);
    const local = git(['config', '--local', '--list']).toString('utf8');
    expect(local).not.toContain('autocrlf');
    expect(local).not.toContain('longpaths');
  });
});
