    22	function startInstaller(script: string, home: string, fakeBin: string, serverUrl: string, args: string[] = [], extraEnv: Record<string, string> = {}, withService = false) {
    23	  let finish!: (result: { code: number | string; output: string }) => void;
    24	  const result = new Promise<{ code: number | string; output: string }>((resolve) => { finish = resolve; });
    25	  const env: NodeJS.ProcessEnv = {
    26	    ...process.env, ...extraEnv,
    27	    NODE_OPTIONS: `--require="${join(fakeBin, "node-preload.cjs")}"`,
    28	    TEST_CLI_TRACE: join(home, "cli-trace.jsonl"),
    29	    TEST_TASK_TRACE: join(home, "task-trace.txt")
    30	  };
    31	  for (const key of Object.keys(env)) {
    32	    if (key.toLowerCase() === "path") delete env[key];
    33	  }
    34	  env.Path = `${fakeBin};${process.env.Path ?? process.env.PATH}`;
    35	  const child = execFile("powershell.exe", [
    36	    "-NoProfile", "-NonInteractive", "-ExecutionPolicy", "Bypass", "-File", script,
    37	    "-ServerUrl", serverUrl, "-ProjectRoot", home, ...(withService ? [] : ["-NoService"]), ...args
    38	  ], {
    39	    env,
    40	    timeout: 90_000,
    41	    maxBuffer: 2 * 1024 * 1024
    42	  }, (error, stdout, stderr) => {
    43	    finish({ code: error ? (error.code ?? 1) : 0, output: stdout + stderr });
    44	  });
    45	  return { child, result };
    46	}
