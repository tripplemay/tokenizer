    48	# $ErrorActionPreference = "Stop" only covers PowerShell's own terminating
    49	# errors — a native executable exiting non-zero sails straight past it. Without
    50	# this wrapper a failed `npm ci` would keep going and still report success,
    51	# which is the behaviour `set -euo pipefail` gives install.sh for free.
    52	function Invoke-Checked {
    53	  param([Parameter(Mandatory)][string] $Exe, [Parameter(ValueFromRemainingArguments)][string[]] $Arguments)
    54	  if ($Arguments -contains "--enroll-token") {
    55	    # Native diagnostics may echo argv. Suppress all streams for enrollment;
    56	    # neither exceptions nor the console may repeat the one-time token.
    57	    $previousPreference = $ErrorActionPreference
    58	    $ErrorActionPreference = "Continue"
    59	    try {
    60	      & $Exe @Arguments *> $null
    61	      $exitCode = $LASTEXITCODE
    62	    } catch {
    63	      throw "Sensitive command failed to start: $Exe"
    64	    } finally {
    65	      $ErrorActionPreference = $previousPreference
    66	    }
    67	  } else {
    68	    & $Exe @Arguments
    69	    $exitCode = $LASTEXITCODE
    70	  }
    71	  if ($exitCode -ne 0) {
    72	    throw "Command failed with exit code ${exitCode}: $Exe"
    73	  }
    74	}
   135	function New-CmdShim {
   136	  # pushd/popd because tsx is resolved relative to the working directory —
   137	  # the same constraint that makes bin/tokenizer set cwd on POSIX.
   138	  # The install path is deliberately NOT interpolated here. It contains the
   139	  # username, which on Windows may be non-ASCII (CJK, Cyrillic, accented
   140	  # Latin); writing it into a .cmd would corrupt it under any single-byte
   141	  # encoding. %~dp0 resolves relative to the shim at runtime instead, so the
   142	  # file stays pure ASCII whatever the user is called.
   143	  $shim = @'
   144	@echo off
   145	pushd "%~dp0..\app"
   146	node --import tsx "src\cli\index.ts" %*
   147	set TOKENIZER_EXIT=%ERRORLEVEL%
   148	popd
   149	exit /b %TOKENIZER_EXIT%
   150	'@
   151	  Set-Content -Path (Join-Path $BinDir "tokenizer.cmd") -Value $shim -Encoding ASCII
   152	}

   172	  $lockStream = [IO.File]::Open($lockFile, [IO.FileMode]::OpenOrCreate, [IO.FileAccess]::ReadWrite, [IO.FileShare]::None)
