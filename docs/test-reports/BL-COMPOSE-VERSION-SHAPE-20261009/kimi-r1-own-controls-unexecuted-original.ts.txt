function atBase(path: string): string {
  const result = spawnSync('git', ['show', `${BASE}:${path}`], { encoding: 'utf8' });
  if (result.status !== 0) throw new Error(`git show ${BASE}:${path}: ${result.stderr}`);
  return result.stdout;
}

function workflowValidator(source = workflowNow) {
  const open = "          if ! jq -e --arg source \"$GITHUB_SHA\" '\n";
  const close = "          ' \"$work/inventory.json\" >/dev/null 2>&1; then";
  const start = source.indexOf(open) + open.length;
  const end = source.indexOf(close, start);
  if (start < open.length || end < start) throw new Error('real workflow validator not found');
  return { jq: source.slice(start, end), outside: `${source.slice(0, start)}<VALIDATOR/>${source.slice(end)}` };
}

// Runs the real workflow jq filter against a candidate report body.
function acceptedByRealValidator(body: unknown): boolean {
  const result = spawnSync('jq', ['-e', '--arg', 'source', SHA, workflowValidator().jq], {
    input: JSON.stringify(body), encoding: 'utf8', env: { PATH: '/usr/bin:/bin', NODE_ENV: 'test' }
  });
  if (result.status !== 0 && result.status !== 1) throw new Error(`jq failed: ${result.stderr}`);
  return result.status === 0;
}

type Shape = {
  prefix: string; core_present: boolean; tail_bytes: number | null; tail_charset: string;
  has_cr: boolean; has_lf: boolean; has_horizontal_space: boolean; has_escape: boolean; has_non_ascii: boolean;
};
const expectShape = (prefix: string, core = false, tailBytes: number | null = null, charset = 'not_parsed', flags: Partial<Shape> = {}): Shape => ({
  prefix, core_present: core, tail_bytes: tailBytes, tail_charset: charset,
  has_cr: false, has_lf: false, has_horizontal_space: false, has_escape: false, has_non_ascii: false, ...flags
});

    const locales = existsSync(lclog) ? readFileSync(lclog, 'utf8').trim().split('\n').map((line) => JSON.parse(line)) : [];
    return { status: result.status, body, calls, locales, elapsedMs };

describe('validator controls: the real workflow jq filter', () => {
  // A synthetically constructed, internally consistent report per probe class,
  // accepted by the real validator before mutation.
  const shape = (prefix: string, core: boolean, tail: number | null, charset: string): Shape => expectShape(prefix, core, tail, charset);
  const fullReport = (probe: Record<string, unknown>, shapeValue: unknown, compose = 'unknown') => ({

  });

describe('mechanical freeze controls: everything outside the permitted scope', () => {
  it('limits the whole-tree diff vs base to the four scoped paths, the spec and its evidence', () => {

    for (const line of lines) {
